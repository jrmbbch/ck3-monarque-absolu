namespace = mqa_decision

# Execute confirmed actions outside the decision tooltip effect tree.
# All eight events are immediate and one-shot. Existing council maintenance is unchanged.

# mqa_money_decision
mqa_decision.1 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "MQA_DECISION BEGIN mqa_money_decision"
		add_gold = 100000
		add_prestige = 100000
		add_piety = 100000
		dynasty ?= {
			add_dynasty_prestige = 100000
		}
		# Resources added by DLCs, only when the character actually uses them
		if = {
			limit = { has_legitimacy = yes }
			add_legitimacy = 100000
		}
		if = {
			limit = { government_has_flag = government_has_influence }	# Roads to Power
			change_influence = 100000
		}
		if = {
			limit = { government_has_flag = government_has_merit }		# All Under Heaven
			change_merit = 100000
		}
		if = {
			limit = { has_treasury = yes }									# administrative and celestial realms
			add_treasury = 100000
		}
		if = {
			limit = { has_royal_court = yes }								# Royal Court
			change_current_court_grandeur = 100
		}
		if = {
			limit = { has_pam_dlc_trigger = yes }							# By God Alone
			change_spiritual_fulfillment = 100
		}
		domicile ?= {
			if = {
				limit = { is_domicile_type = yurt }							# Khans of the Steppe
				change_herd = 100000
			}
			if = {
				limit = { is_domicile_type = camp }							# landless adventurers
				change_provisions = 100000
			}
		}
		debug_log = "MQA_DECISION END mqa_money_decision"
	}
}

# mqa_convert_subrealm_to_culture_decision
mqa_decision.2 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "MQA_DECISION BEGIN mqa_convert_subrealm_to_culture_decision"
		every_sub_realm_county = {
			limit = { NOT = { culture = root.culture } }
			set_county_culture = root.culture
		}
		every_vassal_or_below = {
			limit = { NOT = { culture = root.culture } }
			set_culture = root.culture
		}
		every_child = {
			limit = { NOT = { culture = root.culture} }
			set_culture = root.culture
		}
		every_consort = {
			limit = { NOT = { culture = root.culture } }
			set_culture = root.culture
		}
		every_close_family_member = {
			limit = { NOT = { culture = root.culture } }
			set_culture = root.culture
		}
		every_heir = {
			limit = { NOT = { culture = root.culture } }
			set_culture = root.culture
		}
		house = {
			every_house_member = {
				limit = {
					is_alive = yes
					NOT = { culture = root.culture }
				}
				set_culture = root.culture
			}
		}
		debug_log = "MQA_DECISION END mqa_convert_subrealm_to_culture_decision"
	}
}

# mqa_convert_subrealm_to_religion_decision
mqa_decision.3 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "MQA_DECISION BEGIN mqa_convert_subrealm_to_religion_decision"
		# Faith and rite (see common/scripted_effects/mqa_faith_effects.txt)
		every_sub_realm_county = { mqa_convert_county_to_root_faith_effect = yes }
		every_vassal_or_below = { mqa_convert_character_to_root_faith_effect = yes }
		every_child = {
			limit = { is_alive = yes }
			mqa_convert_character_to_root_faith_effect = yes
		}
		every_consort = { mqa_convert_character_to_root_faith_effect = yes }
		house = {
			every_house_member = {
				limit = { is_alive = yes }
				mqa_convert_character_to_root_faith_effect = yes
			}
		}
		debug_log = "MQA_DECISION END mqa_convert_subrealm_to_religion_decision"
	}
}

# mqa_gather_realm_decision
mqa_decision.4 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "MQA_DECISION BEGIN mqa_gather_realm_decision"
		custom_tooltip = mqa_gather_realm_decision_effect_tt
		hidden_effect = {
			create_title_and_vassal_change = {
				type = conquest
				save_scope_as = change
				add_claim_on_loss = no
			}
			every_held_title = {
				limit = { tier >= tier_duchy }
				every_in_de_jure_hierarchy = {
					limit = {
						tier >= tier_county
						tier <= tier_kingdom
						exists = holder
						holder = {
							NOR = {
								this = root
								target_is_liege_or_above = root
							}
						}
					}
					if = {
						limit = { tier = tier_county }
						change_title_holder_include_vassals = {
							holder = root
							change = scope:change
						}
					}
					else = {
						change_title_holder = {
							holder = root
							change = scope:change
						}
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		debug_log = "MQA_DECISION END mqa_gather_realm_decision"
	}
}
