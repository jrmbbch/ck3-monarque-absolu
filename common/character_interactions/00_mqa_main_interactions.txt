# Based on the vanilla debug interaction of 1.20 (common/character_interactions/00_test_interactions.txt),
# without debug_only: pick any title of the character, landless and administrative titles
# included, either alone or with its vassals.
mqa_take_title_interaction = {
	interface_priority = 10
	category = interaction_debug_pam
	desc = mqa_take_title_interaction_desc
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	target_type = title
	target_filter = recipient_domain_titles

	icon = debug_take_title

	is_shown = {
		scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
		scope:recipient != scope:actor
	}

	can_be_picked_title = {
		scope:target = { is_noble_family_title = no }
	}

	send_options_exclusive = yes
	send_option = {
		flag = title_only
		localization = mqa_take_title_only_option
	}
	send_option = {
		flag = title_and_vassals
		localization = mqa_take_title_and_vassals_option
	}

	auto_accept = yes

	on_accept = {
		# The game moves the de jure vassals of a title along with it (as with a usurpation):
		# remember the holder's vassals so that "title only" can hand them back afterwards.
		if = {
			limit = { scope:title_only = yes }
			scope:recipient = {
				every_vassal = { add_to_temporary_list = mqa_previous_vassals }
			}
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:target = {
			if = {
				limit = { scope:title_and_vassals = yes }
				change_title_holder_include_vassals = {
					holder = scope:actor
					change = scope:change
				}
			}
			else = {
				change_title_holder = {
					holder = scope:actor
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		if = {
			limit = {
				scope:title_only = yes
				scope:recipient = { is_landed = yes }
			}
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = mqa_return_change
			}
			# Only back to a lord who still outranks them (a duke left with counties only
			# cannot keep counts as vassals).
			every_in_list = {
				list = mqa_previous_vassals
				limit = {
					liege ?= scope:actor
					highest_held_title_tier < scope:recipient.highest_held_title_tier
				}
				change_liege = {
					liege = scope:recipient
					change = scope:mqa_return_change
				}
			}
			resolve_title_and_vassal_change = scope:mqa_return_change
		}
	}
}

# Based on the vanilla debug take_realm_interaction of 1.20, without debug_only: take every
# title the character holds (domain, their vassals swear fealty to you), or every title of
# their whole realm, their vassals' counties and duchies included (vassals keep only baronies).
mqa_take_realm_interaction = {
	interface_priority = 20
	category = interaction_debug_pam
	desc = mqa_take_realm_interaction_desc
	common_interaction = yes

	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	icon = grant_titles_interaction

	is_shown = {
		scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
		scope:recipient != scope:actor
	}

	auto_accept = yes

	send_options_exclusive = yes
	send_option = {
		flag = domain
		localization = mqa_take_domain_option
		starts_enabled = { always = no }
	}
	send_option = {
		flag = realm
		localization = mqa_take_realm_option
		starts_enabled = { always = yes }
	}

	on_accept = {
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
		}
		scope:recipient = {
			if = {
				limit = { scope:realm = yes }
				every_sub_realm_title = {
					limit = {
						tier >= tier_county
						is_noble_family_title = no
						is_landless_type_title = no
						NOT = { holder ?= scope:actor }
					}
					change_title_holder_include_vassals = {
						holder = scope:actor
						change = scope:change
						take_baronies = no
					}
				}
			}
			else = {
				every_held_title = {
					limit = {
						tier >= tier_county
						is_noble_family_title = no
						is_landless_type_title = no
					}
					change_title_holder_include_vassals = {
						holder = scope:actor
						change = scope:change
						take_baronies = no
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
}

# Grant the actor pressed claims without transferring titles or vassals.
# Match the title scope of mqa_take_realm_interaction; default to the whole realm.
mqa_claim_realm_interaction = {
	interface_priority = 21
	category = interaction_debug_pam
	desc = mqa_claim_realm_interaction_desc
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = icon_claim_pressed

	is_shown = {
		scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
		scope:recipient != scope:actor
	}

	auto_accept = yes

	send_options_exclusive = yes
	send_option = {
		flag = domain
		localization = mqa_claim_domain_option
		starts_enabled = { always = no }
	}
	send_option = {
		flag = realm
		localization = mqa_claim_realm_option
		starts_enabled = { always = yes }
	}

	on_accept = {
		scope:recipient = {
			if = {
				limit = { scope:realm = yes }
				every_sub_realm_title = {
					limit = {
						tier >= tier_county
						is_noble_family_title = no
						is_landless_type_title = no
						NOT = { holder ?= scope:actor }
					}
					save_temporary_scope_as = mqa_claim_title
					scope:actor = { add_pressed_claim = scope:mqa_claim_title }
				}
			}
			else = {
				every_held_title = {
					limit = {
						tier >= tier_county
						is_noble_family_title = no
						is_landless_type_title = no
						NOT = { holder ?= scope:actor }
					}
					save_temporary_scope_as = mqa_claim_title
					scope:actor = { add_pressed_claim = scope:mqa_claim_title }
				}
			}
		}
	}
}
