mqa_immortal_interaction = {
	icon = debug_good
	category = interaction_mqa_main
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes

	desc = mqa_immortal_interaction_desc
	
	is_shown = {
		scope:recipient = {
			NOT = {
				has_trait = mqa_immortality
			}
		}
	
	}
	on_accept = {
		scope:recipient = {
			add_trait = mqa_immortality
			set_immortal_age = 25
		}
	}
	
	auto_accept = yes
}

mqa_all_in_one_interaction = {
	icon = coronation_icon
	category = interaction_mqa_main
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	
	desc = mqa_all_in_one_interaction_desc

	is_shown = {
		scope:recipient = {
			NOT = {
				AND = {
					has_trait = mqa_super
					has_trait = mqa_super_general
					has_trait = mqa_super_governor										
				}
			}
		}		
	}
	
	on_accept = {
		scope:recipient = {
			add_trait = mqa_super
			add_trait = mqa_super_general
			add_trait = mqa_super_governor			
		}
	}
	auto_accept = yes
}

# 1000 experience per perk point and 27 perks per lifestyle: 30000 unlocks every perk.
mqa_lifestyle_xp_interaction = {
	category = interaction_mqa_main
	common_interaction = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = inspiration

	desc = mqa_lifestyle_xp_interaction_desc

	on_accept = {
		scope:recipient = {
			add_diplomacy_lifestyle_xp = 30000
			add_martial_lifestyle_xp = 30000
			add_stewardship_lifestyle_xp = 30000
			add_intrigue_lifestyle_xp = 30000
			add_learning_lifestyle_xp = 30000
			if = {
				limit = { has_dlc_feature = wandering_nobles }
				add_wanderer_lifestyle_xp = 30000
			}
		}
	}

	auto_accept = yes
}
