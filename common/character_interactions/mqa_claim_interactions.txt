# Based on the vanilla test interactions get_claim_interaction and claim_all_interaction
# of 1.20, without debug_only: a pressed claim on one title of the character, whatever its
# tier, or on every title they hold from county up.

mqa_claim_title_interaction = {
	interface_priority = 26
	category = interaction_debug_pam
	desc = mqa_claim_title_interaction_desc
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = vassal_claim_liege_title_interaction

	target_type = title
	target_filter = recipient_domain_titles

	is_shown = {
		scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
		scope:recipient != scope:actor
	}

	can_be_picked_title = {
		scope:target = {
			is_noble_family_title = no
			is_landless_type_title = no
		}
		# A pressed claim already there: nothing to add.
		trigger_if = {
			limit = { scope:actor = { has_claim_on = scope:target } }
			scope:actor = { has_weak_claim_on = scope:target }
		}
	}

	auto_accept = yes

	on_accept = {
		scope:actor = { add_pressed_claim = scope:target }
	}
}

mqa_claim_all_titles_interaction = {
	interface_priority = 25
	category = interaction_debug_pam
	desc = mqa_claim_all_titles_interaction_desc
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = icon_claim_pressed

	is_shown = {
		scope:recipient = { is_landed_or_landless_administrative_or_theocratic = yes }
		scope:recipient != scope:actor
	}

	auto_accept = yes

	on_accept = {
		scope:recipient = {
			every_held_title = {
				limit = {
					tier >= tier_county
					is_landless_type_title = no
					is_noble_family_title = no
					NAND = {
						scope:actor = { has_claim_on = prev }
						NOT = { scope:actor = { has_weak_claim_on = prev } }
					}
				}
				save_temporary_scope_as = mqa_claimed_title
				scope:actor = { add_pressed_claim = scope:mqa_claimed_title }
			}
		}
	}
}
