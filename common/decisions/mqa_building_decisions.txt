## Monarque Absolu building decisions (mqa_b_ prefix).
## The province decision acts on every holding of the counties the player holds directly,
## vassal barons' cities, temples and castles included: they pay taxes and levies to the player.
## No decision adds building slots: only the slots the game grants are used.
## The heavy lifting is done by common/scripted_effects/mqa_building_effects.txt,
## generated from the game files by tools/generate_building_effects.ps1.

mqa_b_lucky_7_sub_realm_decision = {
	sort_order = 280
	decision_group_type = mqa
	picture = { reference = gfx/interface/illustrations/decisions/decision_realm.dds }
	title = mqa_b_lucky_7_sub_realm_decision_title
	desc = mqa_b_lucky_7_sub_realm_decision_desc
	selection_tooltip = mqa_b_lucky_7_sub_realm_decision_tooltip
	confirm_text = mqa_b_lucky_7_sub_realm_decision_ct

	is_shown = {
		is_ai = no
		has_character_flag = mqa_b_show_decision_flag
	}

	# Fills the existing slots by priority (fortification, levies, then gold), replacing
	# lower-ranked buildings, then upgrades everything as far as the game allows.
	effect = {
		custom_tooltip = mqa_b_domain_effect_tt
		# Only the event is triggered on confirmation; its body is not a tooltip effect.
		hidden_effect = { trigger_event = mqa_building.1 }
	}

	ai_will_do = { base = 0 }
	ai_check_interval = 0
}
