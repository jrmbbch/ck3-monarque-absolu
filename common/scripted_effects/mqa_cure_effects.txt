### Monarque Absolu cures, used by the cure decisions of common/decisions/mqa_decisions.txt

## Who gets cured

# Character scope. Runs $EFFECT$ on the character, their spouses, children and grandchildren.
mqa_every_close_family_effect = {
	$EFFECT$ = yes
	every_consort = { $EFFECT$ = yes }
	every_child = {
		limit = { is_alive = yes }
		$EFFECT$ = yes
		every_child = {
			limit = { is_alive = yes }
			$EFFECT$ = yes
		}
	}
}

# Character scope. Runs $EFFECT$ on the whole realm: the character, their family and
# house, their court and every vassal below them.
mqa_every_realm_character_effect = {
	mqa_every_close_family_effect = { EFFECT = $EFFECT$ }
	house ?= {
		every_house_member = {
			limit = { is_alive = yes }
			$EFFECT$ = yes
		}
	}
	every_courtier = { $EFFECT$ = yes }
	every_vassal_or_below = { $EFFECT$ = yes }
}
